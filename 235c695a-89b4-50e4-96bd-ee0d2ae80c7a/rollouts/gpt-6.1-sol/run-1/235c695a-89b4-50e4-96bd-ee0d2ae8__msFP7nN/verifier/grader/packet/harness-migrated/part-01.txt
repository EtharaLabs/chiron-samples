   22|     YOSYS = ROOT_DIR / "build" / "yosys"
   23| assert YOSYS.exists(), f"can't find yosys executable in {YOSYS}"
   24| 
   25| raise_error = False
   26| # get all macro commands being used
   27| for macro in MACRO_SOURCE.glob("*.ys"):
   28|     # log current test
   29|     relative_path = macro.relative_to(ROOT_DIR)
   30|     logging.log(logging.INFO, f"Checking {relative_path}")
   31|     # files in macros_commands should be named {command}.ys
   32|     command = macro.stem
   33|     # run `yosys -h {command}` to capture current sub-commands
   34|     proc = subprocess.run([YOSYS, "-QTh", command], capture_output=True, text=True)
   35|     with open(macro) as f:
   36|         # {command.ys} starts with two commented lines, the first is the text
   37|         # immediately prior to the macro body.  The second is the text
   38|         # immediately after.
   39|         start = f.readline()
   40|         end = f.readline()
   41|         file_content = f.readlines()
   42|     expected_content = []
   43|     for line in file_content:
   44|         line = line.split("#")[0].strip()
   45|         if line:
   46|             expected_content.append(line)
   47|     # parse {command.ys}
   48|     if "#start:" not in start or "#end:" not in end:
   49|         logging.error(f"Missing start and/or end string in {relative_path}, see {THIS_FILE}")
   50|         raise_error = True
   51|         continue
   52|     start = start.replace("#start:", "").strip()
   53|     end = end.replace("#end:", "").strip()
   54|     # attempt to find macro body in help output
   55|     match = re.fullmatch(f".*{start}(.*){end}.*", proc.stdout, re.DOTALL)
   56|     if not match:
   57|         logging.error(f"Couldn't find {start!r} and/or {end!r} in `yosys -h {command}` output")
   58|         raise_error = True
   59|         continue
   60|     match_content = match.group(1).strip().splitlines()
   61|     actual_content = []
   62|     for line in match_content:
   63|         if line:
   64|             actual_content.append(line)
   65|     # iterate over and compare expected v actual
   66|     for (expected, actual) in zip(expected_content, actual_content):
   67|         expected = expected.strip()
   68|         actual = actual.strip()
   69|         # raw match
   70|         if expected == actual:
   71|             continue
   72| 
   73|         # rip apart formatting to match line parts
   74|         pattern = r"(?P<cmd>\S+)(?P<pass> \[.*\])?(?P<opt>.*?)(?P<cond>\s+\(.*\))?"

   76|             expected_dict = re.fullmatch(pattern, expected).groupdict()

   79|             raise_error = True
   80|             continue
   81| 
   82|         try:
