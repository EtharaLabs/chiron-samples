find_package(GTest REQUIRED)
file(GLOB_RECURSE unit_sources CONFIGURE_DEPENDS "*Test.cc")
foreach(source IN LISTS unit_sources)
  get_filename_component(name "${source}" NAME_WE)
  add_executable(${name} "${source}")
  target_compile_definitions(${name} PRIVATE ${YOSYS_DEFINITIONS})
  target_compile_options(${name} PRIVATE ${YOSYS_OPTIONS})
  target_include_directories(${name} PRIVATE ${YOSYS_INCLUDES})
  target_link_options(${name} PRIVATE ${YOSYS_LINK_OPTIONS})
  target_link_libraries(${name} PRIVATE libyosys GTest::gtest GTest::gmock GTest::gtest_main ${YOSYS_LIBRARIES})
  add_test(NAME ${name} COMMAND ${name})
endforeach()
