foreach(feature READLINE PLUGINS ZLIB TCL)
  if(NOT YOSYS_WITHOUT_${feature})
    list(APPEND YOSYS_DEFINITIONS YOSYS_ENABLE_${feature})
    if(feature STREQUAL "READLINE")
      find_path(READLINE_INCLUDE_DIR readline/readline.h REQUIRED)
      find_library(READLINE_LIBRARY readline REQUIRED)
      list(APPEND YOSYS_INCLUDES ${READLINE_INCLUDE_DIR})
      list(APPEND YOSYS_LIBRARIES ${READLINE_LIBRARY})
      string(APPEND LIBS " -lreadline")
    elseif(feature STREQUAL "ZLIB")
      find_package(ZLIB REQUIRED)
      list(APPEND YOSYS_LIBRARIES ZLIB::ZLIB)
      string(APPEND LIBS " -lz")
    else()
      if(feature STREQUAL "PLUGINS")
        set(pkg libffi)
      else()
        set(pkg tcl)
      endif()
      pkg_check_modules(${feature} REQUIRED ${pkg})
      list(APPEND YOSYS_INCLUDES ${${feature}_INCLUDE_DIRS})
      list(APPEND YOSYS_OPTIONS ${${feature}_CFLAGS_OTHER})
      list(APPEND YOSYS_LIBRARIES ${${feature}_LDFLAGS})
      execute_process(COMMAND ${PKG_CONFIG_EXECUTABLE} --cflags ${pkg} OUTPUT_VARIABLE pkg_cflags OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
      string(APPEND CONFIG_INCLUDES " ${pkg_cflags}")
      execute_process(COMMAND ${PKG_CONFIG_EXECUTABLE} --libs ${pkg} OUTPUT_VARIABLE pkg_libs OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
      string(APPEND LIBS " ${pkg_libs}")
      if(feature STREQUAL "PLUGINS")
        list(APPEND YOSYS_LIBRARIES ${CMAKE_DL_LIBS})
        if(CMAKE_DL_LIBS)
          string(APPEND LIBS " -ldl")
        endif()
      endif()
    endif()
  endif()
endforeach()

function(yosys_pattern output name)
  get_filename_component(dir "${PROJECT_BINARY_DIR}/${output}" DIRECTORY)
  file(MAKE_DIRECTORY "${dir}")
  set(inputs ${ARGN})
  list(TRANSFORM inputs PREPEND "${PROJECT_SOURCE_DIR}/")
  add_custom_command(OUTPUT "${output}" COMMAND ${Python3_EXECUTABLE} "${PROJECT_SOURCE_DIR}/passes/pmgen/pmgen.py" -o "${output}" -p "${name}" ${inputs} DEPENDS passes/pmgen/pmgen.py ${inputs} VERBATIM)
  set(GENERATED ${GENERATED} "${PROJECT_BINARY_DIR}/${output}" PARENT_SCOPE)
endfunction()

function(yosys_data source destination)
  if(YOSYS_WITHOUT_ZLIB AND source MATCHES "^(libs/fst/|kernel/fstdata)")
    return()
  endif()
  if(source MATCHES "(lut_tree_cells.genlib|lut_tree_map.v|bram_types_sim.v)$")
    set(input "${PROJECT_BINARY_DIR}/${source}")
  else()
    set(input "${PROJECT_SOURCE_DIR}/${source}")
  endif()
  get_filename_component(dir "${destination}" DIRECTORY)
  add_custom_command(OUTPUT "${destination}" COMMAND ${CMAKE_COMMAND} -DINPUT=${input} -DOUTPUT=${PROJECT_BINARY_DIR}/${destination} -P "${PROJECT_SOURCE_DIR}/cmake/CopyData.cmake" DEPENDS "${input}" cmake/CopyData.cmake VERBATIM)
  set(DATA_FILES ${DATA_FILES} "${destination}" PARENT_SCOPE)
endfunction()
