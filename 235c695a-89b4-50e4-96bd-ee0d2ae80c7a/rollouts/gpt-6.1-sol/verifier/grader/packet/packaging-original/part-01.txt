   34| class libyosys_so_ext(Extension):
   35|     def __init__(
   36|         self,
   37|     ) -> None:
   38|         super().__init__(
   39|             "libyosys.so",
   40|             [],
   41|         )
   42| 
   43|         # when iterating locally, you probably want to set this variable

   45|         pybind_include = os.getenv("_FORCE_PYBIND_INCLUDE", pybind11.get_include())

   56|             "ABCEXTERNAL=",
   57|             # Show compile commands
   58|             "PRETTY=0",
   59|         ]

   61|     def custom_build(self, bext: build_ext):
   62|         make_flags_split = shlex.split(os.getenv("makeFlags", ""))
   63|         # abc linking takes a lot of memory, best get it out of the way first
   64|         bext.spawn(
   65|             [
   66|                 "make",
   67|                 f"-j{os.cpu_count() or 1}",
   68|                 "yosys-abc",
   69|                 *make_flags_split,
   70|                 *self.args,
   71|             ]
   72|         )
   73|         # build libyosys and share with abc out of the way
   74|         bext.spawn(
   75|             [
   76|                 "make",
   77|                 f"-j{os.cpu_count() or 1}",
   78|                 self.name,
   79|                 "share",
   80|                 *make_flags_split,
   81|                 *self.args,
   82|             ]
   83|         )
   84|         ext_fullpath = Path(bext.get_ext_fullpath(self.name))
   85|         build_path = ext_fullpath.parents[1]
   86|         pyosys_path = build_path / "pyosys"
   87|         os.makedirs(pyosys_path, exist_ok=True)
   88| 
   89|         # libyosys.so
   90|         target = pyosys_path / self.name
   91|         shutil.copy(self.name, target)
   92|         bext.spawn(["strip", "-S", str(target)])
   93| 
   94|         # yosys-abc
   95|         yosys_abc_target = pyosys_path / "yosys-abc"
   96|         shutil.copy("yosys-abc", yosys_abc_target)
   97|         bext.spawn(["strip", "-S", str(yosys_abc_target)])
   98| 
   99|         # share directory
  100|         share_target = pyosys_path / "share"
  101|         try:
  102|             shutil.rmtree(share_target)
  103|         except FileNotFoundError:
  104|             pass
  105| 
  106|         shutil.copytree("share", share_target)
  107| 
  108| 
  109| class custom_build_ext(build_ext):
  110|     def build_extension(self, ext) -> None:
  111|         if not hasattr(ext, "custom_build"):
  112|             return super().build_extension(ext)
  113|         return ext.custom_build(self)
