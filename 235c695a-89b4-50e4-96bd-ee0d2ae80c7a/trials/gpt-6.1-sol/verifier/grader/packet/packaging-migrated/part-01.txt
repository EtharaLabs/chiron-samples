   34| class libyosys_so_ext(Extension):
   35|     def __init__(
   36|         self,
   37|     ) -> None:
   38|         super().__init__(
   39|             "libyosys.so",
   40|             [],
   41|         )

   43|         # when iterating locally, you probably want to set this variable

   45|         pybind_include = os.getenv("_FORCE_PYBIND_INCLUDE", pybind11.get_include())

   48|             "-DYOSYS_ENABLE_PYOSYS=ON",

   51|             "-DBUILD_TESTING=OFF",
   52|         ]

   54|     def custom_build(self, bext: build_ext):
   55|         build_dir = Path(bext.build_temp).resolve() / "yosys"
   56|         cpus = len(os.sched_getaffinity(0)) if hasattr(os, "sched_getaffinity") else (os.cpu_count() or 1)
   57|         quota = Path("/sys/fs/cgroup/cpu.max")
   58|         if quota.exists():
   59|             limit, period = quota.read_text().split()
   60|             if limit != "max":
   61|                 cpus = min(cpus, max(1, int(limit) // int(period)))
   62|         bext.spawn(["cmake", "-S", str(__yosys_root__.resolve()), "-B", str(build_dir),
   63|                     "-G", "Ninja", f"-DPython3_EXECUTABLE={os.sys.executable}",
   64|                     *self.args, *shlex.split(os.getenv("CMAKE_ARGS", ""))])
   65|         bext.spawn(["cmake", "--build", str(build_dir), "--target", "yosys-abc", "-j" + str(cpus)])
   66|         bext.spawn(["cmake", "--build", str(build_dir), "--target", "libyosys", "share", "-j" + str(cpus)])
   67|         ext_fullpath = Path(bext.get_ext_fullpath(self.name))
   68|         build_path = ext_fullpath.parents[1]
   69|         pyosys_path = build_path / "pyosys"
   70|         os.makedirs(pyosys_path, exist_ok=True)
   71| 
   72|         # libyosys.so
   73|         target = pyosys_path / self.name
   74|         shutil.copy(build_dir / self.name, target)
   75|         bext.spawn(["strip", "-S", str(target)])
   76| 
   77|         # yosys-abc
   78|         yosys_abc_target = pyosys_path / "yosys-abc"
   79|         shutil.copy(build_dir / "yosys-abc", yosys_abc_target)
   80|         bext.spawn(["strip", "-S", str(yosys_abc_target)])
   81| 
   82|         # share directory
   83|         share_target = pyosys_path / "share"
   84|         try:
   85|             shutil.rmtree(share_target)
   86|         except FileNotFoundError:
   87|             pass
   88| 
   89|         shutil.copytree(build_dir / "share", share_target)
   90|

   92| class custom_build_ext(build_ext):
   93|     def build_extension(self, ext) -> None:
   94|         if not hasattr(ext, "custom_build"):
   95|             return super().build_extension(ext)
   96|         return ext.custom_build(self)
