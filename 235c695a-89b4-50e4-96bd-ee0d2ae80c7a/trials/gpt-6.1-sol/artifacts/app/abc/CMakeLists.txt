include(${PROJECT_SOURCE_DIR}/cmake/AbcSources.cmake)
list(TRANSFORM ABC_SOURCES PREPEND "${PROJECT_SOURCE_DIR}/abc/")
set(abc_link_order "")
foreach(extension cc cpp c)
  foreach(source IN LISTS ABC_SOURCES)
    if(source MATCHES "\\.${extension}$")
      list(APPEND abc_link_order "${source}")
    endif()
  endforeach()
endforeach()
set(ABC_SOURCES ${abc_link_order})
set_source_files_properties(${ABC_SOURCES} PROPERTIES LANGUAGE CXX)
if(YOSYS_WASI)
  list(REMOVE_ITEM ABC_SOURCES "${PROJECT_SOURCE_DIR}/abc/src/base/main/main.c")
endif()
add_library(abc_objects OBJECT ${ABC_SOURCES})
set_target_properties(abc_objects PROPERTIES CXX_STANDARD 17 POSITION_INDEPENDENT_CODE ${YOSYS_WASI})
target_include_directories(abc_objects PRIVATE ${PROJECT_SOURCE_DIR}/abc/src)
target_compile_definitions(abc_objects PRIVATE ABC_USE_STDINT_H ABC_NAMESPACE=abc ABC_USE_CUDD=1)
set(ABC_OPTFLAGS -g -O)
if(YOSYS_WASI)
  set(ABC_OPTFLAGS -Os)
elseif(DEFINED ENV{OPTFLAGS})
  separate_arguments(ABC_OPTFLAGS NATIVE_COMMAND "$ENV{OPTFLAGS}")
endif()
target_compile_options(abc_objects PRIVATE ${ABC_OPTFLAGS} -Wall -Wno-unused-function -Wno-write-strings -Wno-sign-compare -fpermissive)
# The C sources are compiled as C++ but retain exceptions and the compiler's
# default dialect, as required by ABC's namespace build.
foreach(src IN LISTS ABC_SOURCES)
  if(src MATCHES "\\.(cc|cpp)$")
    set_property(SOURCE ${src} APPEND PROPERTY COMPILE_OPTIONS -std=c++17 -fno-exceptions)
  else()
    set_property(SOURCE ${src} PROPERTY COMPILE_OPTIONS -std=gnu++17)
  endif()
endforeach()
set(ABC_LIBRARIES m ${CMAKE_DL_LIBS})
if(NOT APPLE AND NOT YOSYS_WASI)
  list(APPEND ABC_LIBRARIES rt)
endif()
if(NOT YOSYS_WITHOUT_READLINE)
  target_compile_definitions(abc_objects PRIVATE ABC_USE_READLINE)
  target_include_directories(abc_objects PRIVATE ${READLINE_INCLUDE_DIR})
  list(PREPEND ABC_LIBRARIES ${READLINE_LIBRARY})
endif()
if(NOT YOSYS_DISABLE_THREADS)
  target_compile_definitions(abc_objects PRIVATE ABC_USE_PTHREADS)
  list(APPEND ABC_LIBRARIES Threads::Threads)
endif()
if(YOSYS_WASI)
  target_compile_definitions(abc_objects PRIVATE _WASI_EMULATED_PROCESS_CLOCKS ABC_NO_DYNAMIC_LINKING ABC_NO_RLIMIT)
else()
  add_executable(yosys-abc $<TARGET_OBJECTS:abc_objects>)
  set_target_properties(yosys-abc PROPERTIES OUTPUT_NAME "${YOSYS_PROGRAM_PREFIX}yosys-abc" RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
  target_link_libraries(yosys-abc PRIVATE ${ABC_LIBRARIES})
endif()
