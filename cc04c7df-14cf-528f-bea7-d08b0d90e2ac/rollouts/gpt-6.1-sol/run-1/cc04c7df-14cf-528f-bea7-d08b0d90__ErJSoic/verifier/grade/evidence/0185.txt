73 +|     @GetMapping(value = "/users/{userId}/{year}/{month}/{day}", produces = MediaType.APPLICATION_JSON_VALUE)
74 +|     @ReadRequired
75 +|     public ResponseEntity<String> getUserHistoricStatsHourly(@PathVariable("userId") final String userId,
76 +|                                                @PathVariable("year") final String year,
77 +|                                                @PathVariable("month") final String month,
78 +|                                                @PathVariable("day") final String day,
79 +|                                                final HttpServletRequest request) {
80 +| 
81 +|         try {
82 +|             final int dayAsInt = parseInt(day);
83 +|             final int monthAsInt = parseInt(month);
84 +|             final int yearAsInt = Year.parse(year).getValue();
85 +| 
86 +|             final YearMonth date = YearMonth.of(yearAsInt, monthAsInt);
87 +|             if (!date.isValidDay(dayAsInt)) {
88 +|                 final String errorMessage = String.format("The day '%s' is not a valid day for %s/%s", day, year, month);
89 +|                 LOGGER.error(errorMessage);
90 +|                 return badRequest(errorMessage);
91 +|             }
92 +| 
93 +|             final IdResult idResult = IntegerParser.parsePositive(userId);
94 +|             if (idResult.isFailure()) {
95 +|                 return idResult.getFailureResponse();
96 +|             }
97 +|             final int parsedId = idResult.getId();
98 +| 
99 +|             final Optional<User> user = foldingStatsCore.getUserWithPasskey(parsedId);
100 +|             if (user.isEmpty()) {
101 +|                 LOGGER.error("No user found with ID: {}", parsedId);
102 +|                 return notFound();
103 +|             }
104 +| 
105 +|             final Collection<HistoricStats> historicStats = foldingStatsCore.getHistoricStats(user.get(), Year.parse(year), Month.of(monthAsInt),
106 +|                 dayAsInt);
107 +|             return cachedOk(historicStats, request, CACHE_EXPIRATION_TIME);
108 +|         } catch (final DateTimeParseException e) {
109 +|             final String errorMessage = String.format("The year '%s' is not a valid format", year);
110 +|             LOGGER.debug(errorMessage, e);
111 +|             LOGGER.error(errorMessage);
112 +|             return badRequest(errorMessage);
113 +|         } catch (final DateTimeException e) {
114 +|             final String errorMessage = String.format("The month '%s' is not a valid format", month);
115 +|             LOGGER.debug(errorMessage, e);
116 +|             LOGGER.error(errorMessage);
117 +|             return badRequest(errorMessage);
118 +|         } catch (final NumberFormatException e) {
119 +|             final String errorMessage = String.format("The month '%s' or day '%s' is not a valid format", month, day);
120 +|             LOGGER.debug(errorMessage, e);
121 +|             LOGGER.error(errorMessage);
122 +|             return badRequest(errorMessage);
123 +|         } catch (final Exception e) {
124 +|             LOGGER.error("Unexpected error getting user with ID: {}", userId, e);
125 +|             return serverError();
126 +|         }
127 +|     }
