25 +| package me.zodac.folding.service.tc.user;
26 +| 
27 +| import org.springframework.stereotype.Service;
28 +| import org.springframework.beans.factory.annotation.Autowired;
29 +| import org.springframework.context.annotation.Lazy;
30 +| 
31 +| import java.util.Optional;
32 +| import me.zodac.folding.api.tc.Team;
33 +| import me.zodac.folding.api.tc.User;
34 +| import me.zodac.folding.service.api.FoldingStatsCore;
35 +| import org.apache.logging.log4j.LogManager;
36 +| import org.apache.logging.log4j.Logger;
37 +| 
38 +| @Service
39 +| public class UserCaptainHandler {
40 +| 
41 +|     private static final Logger LOGGER = LogManager.getLogger();
42 +| 
43 +|     @Autowired
44 +|     @Lazy
45 +|     private FoldingStatsCore foldingStatsCore;
46 +| 
47 +|     public boolean isUserCaptainAndCaptainExistsOnTeam(final User user) {
48 +|         if (!user.isUserIsCaptain()) {
49 +|             return false;
50 +|         }
51 +| 
52 +|         final Team team = user.getTeam();
53 +|         final Optional<User> existingCaptainOptional = getCaptainOfTeam(team);
54 +|         if (existingCaptainOptional.isEmpty()) {
55 +|             return false;
56 +|         }
57 +| 
58 +|         final User existingCaptain = existingCaptainOptional.get();
59 +|         LOGGER.info("Captain '{} (ID: {})' already exists for team '{}', will be replaced by '{}' (ID: {})",
60 +|             existingCaptain.getDisplayName(), existingCaptain.getId(), team.getTeamName(), user.getDisplayName(), user.getId()
61 +|         );
62 +|         return true;
63 +|     }
64 +| 
65 +|     public void removeCaptaincyFromExistingTeamCaptain(final Team team) {
66 +|         final Optional<User> existingCaptainOptional = getCaptainOfTeam(team);
67 +|         if (existingCaptainOptional.isEmpty()) {
68 +|             return;
69 +|         }
70 +| 
71 +|         final User existingCaptain = existingCaptainOptional.get();
72 +|         final User userWithCaptaincyRemoved = User.removeCaptaincyFromUser(existingCaptain);
73 +| 
74 +|         foldingStatsCore.updateUser(userWithCaptaincyRemoved, existingCaptain);
75 +|     }
76 +| 
77 +|     private Optional<User> getCaptainOfTeam(final Team team) {
78 +|         return foldingStatsCore.getUsersOnTeam(team)
79 +|             .stream()
80 +|             .filter(User::isUserIsCaptain)
81 +|             .findAny();
82 +|     }
83 +| }
