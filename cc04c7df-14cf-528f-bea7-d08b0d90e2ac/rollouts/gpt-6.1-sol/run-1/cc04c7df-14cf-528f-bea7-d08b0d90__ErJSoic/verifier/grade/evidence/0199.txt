176 +|     @Override
177 +|     public User createUser(final User user) {
178 +|         if (userCaptainHandler.isUserCaptainAndCaptainExistsOnTeam(user)) {
179 +|             userCaptainHandler.removeCaptaincyFromExistingTeamCaptain(user.getTeam());
180 +|         }
181 +| 
182 +|         final User createdUser = storage.createUser(user);
183 +| 
184 +|         try {
185 +|             // When adding a new user, we configure the initial stats DB/cache
186 +|             final UserStats currentUserStats = foldingStatsRetriever.getTotalStats(createdUser);
187 +|             final UserStats initialStats = createInitialStats(currentUserStats);
188 +|             LOGGER.info("User '{}' (ID: {}) created with initial stats: {}", createdUser.getDisplayName(), createdUser.getId(), initialStats);
189 +|             userStatsParser.parseTcStatsForUser(createdUser);
190 +|         } catch (final ExternalConnectionException e) {
191 +|             LOGGER.error("Error retrieving initial stats for user '{}' (ID: {})", createdUser.getDisplayName(), createdUser.getId(), e);
192 +|         }
193 +| 
194 +|         return createdUser;
195 +|     }

226 +|     @Override
227 +|     public User updateUser(final User userToUpdate, final User existingUser) {
228 +|         if (userCaptainHandler.isUserCaptainAndCaptainExistsOnTeam(userToUpdate)) {
229 +|             final boolean isCaptainChange = userToUpdate.isUserIsCaptain() != existingUser.isUserIsCaptain();
230 +|             final boolean isTeamChange = userToUpdate.getTeam().getId() != existingUser.getTeam().getId();
231 +| 
232 +|             if (isCaptainChange || isTeamChange) {
233 +|                 userCaptainHandler.removeCaptaincyFromExistingTeamCaptain(userToUpdate.getTeam());
234 +|             }
235 +|         }
236 +| 
237 +|         // Perform any stats handling before updating the user
238 +|         if (userTeamChangeHandler.isUserTeamChange(userToUpdate, existingUser)) {
239 +|             userTeamChangeHandler.handleTeamChange(userToUpdate, existingUser.getTeam());
240 +|         }
241 +| 
242 +|         final User updatedUser = storage.updateUser(userToUpdate);
243 +| 
244 +|         if (userStateChangeHandler.isUserStateChange(updatedUser, existingUser)) {
245 +|             userStateChangeHandler.handleStateChange(updatedUser);
246 +|             LOGGER.trace("User updated with required state change");
247 +|         }
248 +| 
249 +|         return updatedUser;
250 +|     }

253 +|     public void deleteUser(final User user) {
254 +|         // Retrieve the user's stats before deleting the user, so we can use the values for the retried user stats
255 +|         final UserTcStats userStats = getHourlyTcStats(user);
256 +|         storage.deleteUser(user.getId());
257 +| 
258 +|         if (userStats.isEmptyStats()) {
259 +|             LOGGER.warn("User '{}' (ID: {}) has no stats, not saving any retired stats", user.getDisplayName(), user.getId());
260 +|             return;
261 +|         }
262 +| 
263 +|         final RetiredUserTcStats retiredUserTcStats = RetiredUserTcStats.createWithoutId(user.getTeam().getId(), user.getDisplayName(), userStats);
264 +|         final RetiredUserTcStats createdRetiredUserTcStats = storage.createRetiredUserStats(retiredUserTcStats);
265 +|         LOGGER.info("User '{}' (ID: {}) retired with retired stats ID: {}", user.getDisplayName(), user.getId(),
266 +|             createdRetiredUserTcStats.getRetiredUserId());
267 +|     }

429 +|

444 +|     private CompetitionSummary createCompetitionSummary() {
445 +|         final List<TeamSummary> teamSummaries = getStatsForTeams();

447 +|

462 +|     private TeamSummary getTcTeamResult(final Team team) {
463 +|         LOGGER.debug("Converting team '{}' for TC stats", team::getTeamName);
464 +| 
465 +|         final Collection<User> usersOnTeam = getUsersOnTeam(team);
466 +| 
467 +|         final Collection<UserSummary> activeUserSummaries = usersOnTeam
468 +|             .stream()
469 +|             .map(this::getTcStatsForUser)
470 +|             .collect(toList());
471 +| 
472 +|         final Collection<RetiredUserSummary> retiredUserSummaries = getAllRetiredUsersForTeam(team)
473 +|             .stream()
474 +|             .map(RetiredUserSummary::createWithDefaultRank)
475 +|             .collect(toList());
476 +| 
477 +|         final String captainDisplayName = getCaptainDisplayName(team.getTeamName(), usersOnTeam);
478 +|         return TeamSummary.createWithDefaultRank(team, captainDisplayName, activeUserSummaries, retiredUserSummaries);
479 +|     }

488 +|     private UserSummary getTcStatsForUser(final User user) {
489 +|         final UserTcStats userTcStats = getHourlyTcStats(user);
