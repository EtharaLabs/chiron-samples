25 +| package me.zodac.folding.rest.endpoint;
26 +| 
27 +| import org.springframework.web.bind.annotation.*;
28 +| import org.springframework.http.ResponseEntity;
29 +| import org.springframework.http.MediaType;
30 +| import org.springframework.beans.factory.annotation.Autowired;
31 +| import org.springframework.context.annotation.Lazy;
32 +| import me.zodac.folding.rest.RequiredRoles;
33 +| 
34 +| import static me.zodac.folding.rest.response.Responses.ok;
35 +| import static me.zodac.folding.rest.response.Responses.serverError;
36 +| 
37 +| import me.zodac.folding.service.api.FoldingStatsCore;
38 +| import me.zodac.folding.service.tc.lars.LarsHardwareUpdater;
39 +| import org.apache.logging.log4j.LogManager;
40 +| import org.apache.logging.log4j.Logger;
41 +| 
42 +| @RestController
43 +| @RequestMapping("/debug")
44 +| public class DebugEndpoint {
45 +| 
46 +|     private static final Logger LOGGER = LogManager.getLogger();
47 +| 
48 +|     @Autowired
49 +|     @Lazy
50 +|     private FoldingStatsCore foldingStatsCore;
51 +| 
52 +|     @Autowired
53 +|     @Lazy
54 +|     private LarsHardwareUpdater larsHardwareUpdater;
55 +| 
56 +|     @PostMapping(value = "/lars", produces = MediaType.APPLICATION_JSON_VALUE)
57 +|     @RequiredRoles("admin")
58 +|     public ResponseEntity<String> startLarsUpdate() {
59 +|         LOGGER.info("GET request received to manually update hardware from LARS DB");
60 +|         try {
61 +|             larsHardwareUpdater.retrieveHardwareAndPersist();
62 +|             return ok();
63 +|         } catch (final Exception e) {
64 +|             LOGGER.error("Unexpected error updating hardware from LARS", e);
65 +|             return serverError();
66 +|         }
67 +|     }
68 +|     @PostMapping(value = "/caches", produces = MediaType.APPLICATION_JSON_VALUE)
69 +|     @RequiredRoles("admin")
70 +|     public ResponseEntity<String> printCaches() {
71 +|         LOGGER.info("Printing cache contents");
72 +|         foldingStatsCore.printCacheContents();
73 +|         return ok();
74 +|     }
75 +| }
