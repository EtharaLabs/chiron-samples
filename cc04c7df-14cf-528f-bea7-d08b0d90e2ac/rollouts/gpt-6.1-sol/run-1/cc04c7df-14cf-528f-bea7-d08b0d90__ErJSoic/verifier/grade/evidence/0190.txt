82 +|     public ResponseEntity<String> create(@RequestBody(required = false) final UserRequest userRequest) {
83 +| 
84 +|         final ValidationResult<User> validationResult = validateCreate(userRequest);
85 +|         if (validationResult.isFailure()) {
86 +|             return ValidationFailureResponseMapper.map(validationResult);
87 +|         }
88 +|         final User validatedUser = validationResult.getOutput();
89 +| 
90 +|         try {
91 +|             final User elementWithId = foldingStatsCore.createUser(validatedUser);
92 +| 
93 +|             final UriComponentsBuilder elementLocationBuilder = ServletUriComponentsBuilder.fromCurrentRequest()
94 +|                 .pathSegment(String.valueOf(elementWithId.getId()));
95 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
96 +|             LOGGER.info("Created user with ID {}", elementWithId.getId());
97 +|             return created(elementWithId, elementLocationBuilder);
98 +|         } catch (final Exception e) {
99 +|             LOGGER.error("Unexpected error creating user: {}", userRequest, e);
100 +|             return serverError();
101 +|         }

103 +|     @PostMapping(value = "/batch", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
104 +|     @WriteRequired
105 +|     @RequiredRoles("admin")
106 +|     public ResponseEntity<String> createBatchOf(@RequestBody(required = false) final Collection<UserRequest> userRequests) {
107 +| 
108 +|         final Collection<User> validUsers = new ArrayList<>(userRequests.size() / 2);
109 +|         final Collection<ValidationFailure> failedValidationResponses = new ArrayList<>(userRequests.size() / 2);
110 +| 
111 +|         for (final UserRequest userRequest : userRequests) {
112 +|             final ValidationResult<User> validationResult = validateCreate(userRequest);
113 +| 
114 +|             if (validationResult.isFailure()) {
115 +|                 LOGGER.error("Found validation error for {}: {}", userRequest, validationResult);
116 +|                 failedValidationResponses.add(validationResult.getValidationFailure());
117 +|             } else {
118 +|                 validUsers.add(validationResult.getOutput());
119 +|             }
120 +|         }
121 +| 
122 +|         if (validUsers.isEmpty()) {
123 +|             LOGGER.error("All users contain validation errors: {}", failedValidationResponses);
124 +|             return badRequest(failedValidationResponses);
125 +|         }
126 +| 
127 +|         final List<Object> successful = new ArrayList<>();
128 +|         final List<Object> unsuccessful = new ArrayList<>(failedValidationResponses);
129 +| 
130 +|         for (final User validUser : validUsers) {
131 +|             try {
132 +|                 final User userWithId = foldingStatsCore.createUser(validUser);
133 +|                 successful.add(userWithId);
134 +|             } catch (final Exception e) {
135 +|                 LOGGER.error("Unexpected error creating user: {}", validUser, e);
136 +|                 unsuccessful.add(validUser);
137 +|             }
138 +|         }
139 +| 
140 +|         final BatchCreateResponse batchCreateResponse = BatchCreateResponse.create(successful, unsuccessful);
141 +| 
142 +|         if (successful.isEmpty()) {
143 +|             return badRequest(batchCreateResponse);
144 +|         }
145 +| 
146 +|         if (!unsuccessful.isEmpty()) {
147 +|             LOGGER.error("{} users successfully created, {} users unsuccessful", successful.size(), unsuccessful.size());
148 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
149 +|             return ok(batchCreateResponse);
150 +|         }
151 +| 
152 +|         LOGGER.info("{} users successfully created", successful.size());
153 +|         SystemStateManager.next(SystemState.WRITE_EXECUTED);
154 +|         return ok(batchCreateResponse.getSuccessful());
155 +|     }

192 +|     @PutMapping(value = "/{userId}", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
193 +|     @WriteRequired
194 +|     @RequiredRoles("admin")
195 +|     public ResponseEntity<String> updateById(@PathVariable("userId") final String userId, @RequestBody(required = false) final UserRequest userRequest) {
196 +| 
197 +|         if (userRequest == null) {
198 +|             LOGGER.error("No payload provided");
199 +|             return nullRequest();
200 +|         }
201 +| 
202 +|         try {
203 +|             final IdResult idResult = IntegerParser.parsePositive(userId);
204 +|             if (idResult.isFailure()) {
205 +|                 return idResult.getFailureResponse();
206 +|             }
207 +|             final int parsedId = idResult.getId();
208 +| 
209 +|             final Optional<User> optionalElement = foldingStatsCore.getUserWithPasskey(parsedId);
210 +|             if (optionalElement.isEmpty()) {
211 +|                 LOGGER.error("No user found with ID {}", userId);
212 +|                 return notFound();
213 +|             }
214 +|             final User existingUser = optionalElement.get();
215 +| 
216 +|             if (existingUser.isEqualRequest(userRequest)) {
217 +|                 LOGGER.debug("No change necessary");
218 +|                 final User userWithHiddenPasskey = User.hidePasskey(existingUser);
219 +|                 return ok(userWithHiddenPasskey);
220 +|             }
221 +| 
222 +|             final ValidationResult<User> validationResult = validateUpdate(userRequest, existingUser);
223 +|             if (validationResult.isFailure()) {
224 +|                 return ValidationFailureResponseMapper.map(validationResult);
225 +|             }
226 +|             final User validatedUser = validationResult.getOutput();
227 +| 
228 +|             // The payload 'should' have the ID, but it's not guaranteed if the correct URL is used
229 +|             final User userWithId = User.updateWithId(existingUser.getId(), validatedUser);
230 +|             final User updatedUserWithId = foldingStatsCore.updateUser(userWithId, existingUser);
231 +| 
232 +|             final UriComponentsBuilder elementLocationBuilder = ServletUriComponentsBuilder.fromCurrentRequest()
233 +|                 .pathSegment(String.valueOf(updatedUserWithId.getId()));
234 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
235 +|             LOGGER.info("Updated user with ID {}", updatedUserWithId.getId());
236 +|             return ok(updatedUserWithId, elementLocationBuilder);
237 +|         } catch (final Exception e) {
238 +|             LOGGER.error("Unexpected error updating user with ID: {}", userId, e);
239 +|             return serverError();
240 +|         }
241 +|     }
