80 +|     @WriteRequired
81 +|     @RequiredRoles("admin")
82 +|     public ResponseEntity<String> create(@RequestBody(required = false) final HardwareRequest hardwareRequest) {
83 +| 
84 +|         final ValidationResult<Hardware> validationResult = validateCreate(hardwareRequest);
85 +|         if (validationResult.isFailure()) {
86 +|             return ValidationFailureResponseMapper.map(validationResult);
87 +|         }
88 +|         final Hardware validatedHardware = validationResult.getOutput();
89 +| 
90 +|         try {
91 +|             final Hardware elementWithId = foldingStatsCore.createHardware(validatedHardware);
92 +| 
93 +|             final UriComponentsBuilder elementLocationBuilder = ServletUriComponentsBuilder.fromCurrentRequest()
94 +|                 .pathSegment(String.valueOf(elementWithId.getId()));
95 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
96 +|             LOGGER.info("Created hardware with ID {}", elementWithId.getId());
97 +|             return created(elementWithId, elementLocationBuilder);
98 +|         } catch (final Exception e) {
99 +|             LOGGER.error("Unexpected error creating hardware: {}", hardwareRequest, e);
100 +|             return serverError();
101 +|         }

103 +|     @PostMapping(value = "/batch", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
104 +|     @WriteRequired
105 +|     @RequiredRoles("admin")
106 +|     public ResponseEntity<String> createBatchOf(@RequestBody(required = false) final Collection<HardwareRequest> hardwareRequests) {
107 +| 
108 +|         final Collection<Hardware> validHardwares = new ArrayList<>(hardwareRequests.size() / 2);
109 +|         final Collection<ValidationFailure> failedValidationResponses = new ArrayList<>(hardwareRequests.size() / 2);
110 +| 
111 +|         for (final HardwareRequest hardwareRequest : hardwareRequests) {
112 +|             final ValidationResult<Hardware> validationResult = validateCreate(hardwareRequest);
113 +| 
114 +|             if (validationResult.isFailure()) {
115 +|                 LOGGER.error("Found validation error for {}: {}", hardwareRequest, validationResult);
116 +|                 failedValidationResponses.add(validationResult.getValidationFailure());
117 +|             } else {
118 +|                 validHardwares.add(validationResult.getOutput());
119 +|             }
120 +|         }
121 +| 
122 +|         if (validHardwares.isEmpty()) {
123 +|             LOGGER.error("All hardwares contain validation errors: {}", failedValidationResponses);
124 +|             return badRequest(failedValidationResponses);
125 +|         }
126 +| 
127 +|         final List<Object> successful = new ArrayList<>();
128 +|         final List<Object> unsuccessful = new ArrayList<>(failedValidationResponses);
129 +| 
130 +|         for (final Hardware validHardware : validHardwares) {
131 +|             try {
132 +|                 final Hardware hardwareWithId = foldingStatsCore.createHardware(validHardware);
133 +|                 successful.add(hardwareWithId);
134 +|             } catch (final Exception e) {
135 +|                 LOGGER.error("Unexpected error creating hardware: {}", validHardware, e);
136 +|                 unsuccessful.add(validHardware);
137 +|             }
138 +|         }
139 +| 
140 +|         final BatchCreateResponse batchCreateResponse = BatchCreateResponse.create(successful, unsuccessful);
141 +| 
142 +|         if (successful.isEmpty()) {
143 +|             return badRequest(batchCreateResponse);
144 +|         }
145 +| 
146 +|         if (!unsuccessful.isEmpty()) {
147 +|             LOGGER.error("{} hardwares successfully created, {} hardwares unsuccessful", successful.size(), unsuccessful.size());
148 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
149 +|             return ok(batchCreateResponse);
150 +|         }
151 +| 
152 +|         LOGGER.info("{} hardwares successfully created", successful.size());
153 +|         SystemStateManager.next(SystemState.WRITE_EXECUTED);
154 +|         return ok(batchCreateResponse.getSuccessful());
155 +|     }

220 +|     @PutMapping(value = "/{hardwareId}", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
221 +|     @WriteRequired
222 +|     @RequiredRoles("admin")
223 +|     public ResponseEntity<String> updateById(@PathVariable("hardwareId") final String hardwareId, @RequestBody(required = false) final HardwareRequest hardwareRequest) {
224 +| 
225 +|         if (hardwareRequest == null) {
226 +|             LOGGER.error("No payload provided");
227 +|             return nullRequest();
228 +|         }
229 +| 
230 +|         try {
231 +|             final IdResult idResult = IntegerParser.parsePositive(hardwareId);
232 +|             if (idResult.isFailure()) {
233 +|                 return idResult.getFailureResponse();
234 +|             }
235 +|             final int parsedId = idResult.getId();
236 +| 
237 +|             final Optional<Hardware> optionalElement = foldingStatsCore.getHardware(parsedId);
238 +|             if (optionalElement.isEmpty()) {
239 +|                 LOGGER.error("No hardware found with ID {}", hardwareId);
240 +|                 return notFound();
241 +|             }
242 +|             final Hardware existingHardware = optionalElement.get();
243 +| 
244 +|             if (existingHardware.isEqualRequest(hardwareRequest)) {
245 +|                 LOGGER.debug("No change necessary");
246 +|                 return ok(existingHardware);
247 +|             }
248 +| 
249 +|             final ValidationResult<Hardware> validationResult = validateUpdate(hardwareRequest, existingHardware);
250 +|             if (validationResult.isFailure()) {
251 +|                 return ValidationFailureResponseMapper.map(validationResult);
252 +|             }
253 +|             final Hardware validatedHardware = validationResult.getOutput();
254 +| 
255 +|             // The payload 'should' have the ID, but it's not guaranteed if the correct URL is used
256 +|             final Hardware hardwareWithId = Hardware.updateWithId(existingHardware.getId(), validatedHardware);
257 +|             final Hardware updatedHardwareWithId = foldingStatsCore.updateHardware(hardwareWithId, existingHardware);
258 +| 
259 +|             final UriComponentsBuilder elementLocationBuilder = ServletUriComponentsBuilder.fromCurrentRequest()
260 +|                 .pathSegment(String.valueOf(updatedHardwareWithId.getId()));
261 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
262 +|             LOGGER.info("Updated hardware with ID {}", updatedHardwareWithId.getId());
263 +|             return ok(updatedHardwareWithId, elementLocationBuilder);
264 +|         } catch (final Exception e) {
265 +|             LOGGER.error("Unexpected error updating hardware with ID: {}", hardwareId, e);
266 +|             return serverError();
267 +|         }
268 +|     }

287 +|
