48 +| @RestController
49 +| @RequestMapping("/login")
50 +| public class LoginEndpoint {
51 +| 
52 +|     private static final Logger LOGGER = LogManager.getLogger();
53 +| 
54 +|     @Autowired
55 +|     @Lazy
56 +|     private FoldingStatsCore foldingStatsCore;
57 +| 
58 +|     @PostMapping(value = "/admin", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
59 +|     @ReadRequired
60 +|     public ResponseEntity<String> loginAsAdmin(@RequestBody(required = false) final LoginCredentials loginCredentials) {
61 +|         LOGGER.debug("Login request received");
62 +| 
63 +|         if (loginCredentials == null || EncodingUtils.isInvalidBasicAuthentication(loginCredentials.getEncodedUserNameAndPassword())) {
64 +|             LOGGER.error("Invalid payload: {}", loginCredentials);
65 +|             return badRequest(loginCredentials);
66 +|         }
67 +| 
68 +|         try {
69 +|             final Map<String, String> decodedUserNameAndPassword =
70 +|                 EncodingUtils.decodeBasicAuthentication(loginCredentials.getEncodedUserNameAndPassword());
71 +|             final String userName = decodedUserNameAndPassword.get(EncodingUtils.DECODED_USERNAME_KEY);
72 +|             final String password = decodedUserNameAndPassword.get(EncodingUtils.DECODED_PASSWORD_KEY);
73 +|             LOGGER.debug("Login request received for user: '{}'", userName);
74 +| 
75 +|             final UserAuthenticationResult userAuthenticationResult = foldingStatsCore.authenticateSystemUser(userName, password);
76 +| 
77 +|             if (!userAuthenticationResult.isUserExists() || !userAuthenticationResult.isPasswordMatch()) {
78 +|                 LOGGER.warn("Invalid user credentials supplied: {}", loginCredentials);
79 +|                 return unauthorized();
80 +|             }
81 +| 
82 +|             if (!userAuthenticationResult.isAdmin()) {
83 +|                 LOGGER.warn("User '{}' is not an admin", userName);
84 +|                 return forbidden();
85 +|             }
86 +| 
87 +|             LOGGER.info("Admin user '{}' logged in", userName);
88 +|             return ok();
89 +|         } catch (final IllegalArgumentException e) {
90 +|             LOGGER.error("Encoded username and password was not a valid Base64 string", e);
91 +|             return badRequest(loginCredentials);
92 +|         } catch (final Exception e) {
93 +|             LOGGER.error("Unexpected error validating user credentials", e);
94 +|             return serverError();
95 +|         }
96 +|     }
97 +| }
