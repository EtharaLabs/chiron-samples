25 +| package me.zodac.folding.service.tc.user;
26 +| 
27 +| import org.springframework.stereotype.Service;
28 +| import org.springframework.beans.factory.annotation.Autowired;
29 +| import org.springframework.context.annotation.Lazy;
30 +| 
31 +| import me.zodac.folding.api.util.ProcessingType;
32 +| import me.zodac.folding.service.api.FoldingStatsCore;
33 +| import me.zodac.folding.service.tc.scheduled.StatsScheduler;
34 +| import org.apache.logging.log4j.LogManager;
35 +| import org.apache.logging.log4j.Logger;
36 +| 
37 +| @Service
38 +| public class UserStatsResetter {
39 +| 
40 +|     private static final Logger LOGGER = LogManager.getLogger();
41 +| 
42 +|     @Autowired
43 +|     @Lazy
44 +|     private FoldingStatsCore foldingStatsCore;
45 +| 
46 +|     @Autowired
47 +|     @Lazy
48 +|     private StatsScheduler statsScheduler;
49 +| 
50 +|     public void resetTeamCompetitionStats() {
51 +|         try {
52 +|             // Pull stats one more time to get the latest values
53 +|             statsScheduler.manualTeamCompetitionStatsParsing(ProcessingType.SYNCHRONOUS);
54 +| 
55 +|             LOGGER.info("Resetting Team Competition stats");
56 +|             foldingStatsCore.resetAllTeamCompetitionUserStats();
57 +| 
58 +|             // Pull stats for new month
59 +|             statsScheduler.manualTeamCompetitionStatsParsing(ProcessingType.SYNCHRONOUS);
60 +|         } catch (final Exception e) {
61 +|             LOGGER.warn("Unexpected error manually resetting TC stats", e);
62 +|         }
63 +|     }
64 +| }
