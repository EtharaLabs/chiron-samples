25 +| package me.zodac.folding.service.tc.user;
26 +| 
27 +| import org.springframework.stereotype.Service;
28 +| import org.springframework.beans.factory.annotation.Autowired;
29 +| import org.springframework.context.annotation.Lazy;
30 +| 
31 +| import me.zodac.folding.api.tc.result.MonthlyResult;
32 +| import me.zodac.folding.service.api.FoldingStatsCore;
33 +| import me.zodac.folding.service.tc.LeaderboardStatsGenerator;
34 +| import org.apache.logging.log4j.LogManager;
35 +| import org.apache.logging.log4j.Logger;
36 +| 
37 +| @Service
38 +| public class UserStatsStorer {
39 +| 
40 +|     private static final Logger LOGGER = LogManager.getLogger();
41 +| 
42 +|     @Autowired
43 +|     @Lazy
44 +|     private FoldingStatsCore foldingStatsCore;
45 +| 
46 +|     @Autowired
47 +|     @Lazy
48 +|     private LeaderboardStatsGenerator leaderboardStatsGenerator;
49 +| 
50 +|     public void storeMonthlyResult() {
51 +|         final MonthlyResult monthlyResult = MonthlyResult.create(
52 +|             leaderboardStatsGenerator.generateTeamLeaderboards(),
53 +|             leaderboardStatsGenerator.generateUserCategoryLeaderboards()
54 +|         );
55 +| 
56 +|         if (monthlyResult.hasNoStats()) {
57 +|             LOGGER.error("Not storing result, result has no stats: {}", monthlyResult);
58 +|             return;
59 +|         }
60 +| 
61 +|         final MonthlyResult createdMonthlyResult = foldingStatsCore.createMonthlyResult(monthlyResult);
62 +|         LOGGER.info("Storing TC results for {}", createdMonthlyResult.getUtcTimestamp());
63 +|     }
64 +| }
