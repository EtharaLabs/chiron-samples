27 +| import org.springframework.web.bind.annotation.*;
28 +| import org.springframework.http.ResponseEntity;
29 +| import org.springframework.http.MediaType;
30 +| import org.springframework.beans.factory.annotation.Autowired;
31 +| import org.springframework.context.annotation.Lazy;
32 +| import me.zodac.folding.rest.RequiredRoles;
33 +| 
34 +| import static me.zodac.folding.rest.response.Responses.badRequest;
35 +| import static me.zodac.folding.rest.response.Responses.ok;
36 +| import static me.zodac.folding.rest.response.Responses.serverError;
37 +| 
38 +| import java.time.DateTimeException;
39 +| import java.time.Month;
40 +| import java.time.Year;
41 +| import java.time.format.DateTimeParseException;
42 +| import java.util.Optional;
43 +| import me.zodac.folding.api.state.ReadRequired;
44 +| import me.zodac.folding.api.state.WriteRequired;
45 +| import me.zodac.folding.api.tc.result.MonthlyResult;
46 +| import me.zodac.folding.service.api.FoldingStatsCore;
47 +| import me.zodac.folding.service.tc.user.UserStatsStorer;
48 +| import org.apache.logging.log4j.LogManager;
49 +| import org.apache.logging.log4j.Logger;
50 +| 
51 +| @RestController
52 +| @RequestMapping("/results")
53 +| public class MonthlyResultEndpoint {
54 +| 
55 +|     private static final Logger LOGGER = LogManager.getLogger();
56 +| 
57 +|     @Autowired
58 +|     @Lazy
59 +|     private FoldingStatsCore foldingStatsCore;
60 +| 
61 +|     @Autowired
62 +|     @Lazy
63 +|     private UserStatsStorer userStatsStorer;
64 +| 
65 +|     @GetMapping(value = "/result/{year}/{month}", produces = MediaType.APPLICATION_JSON_VALUE)
66 +|     @ReadRequired
67 +|     public ResponseEntity<String> getMonthlyResult(@PathVariable("year") final String year,
68 +|                                      @PathVariable("month") final String month) {
69 +| 
70 +|         try {
71 +|             final Optional<MonthlyResult> monthlyResult = foldingStatsCore.getMonthlyResult(Month.of(Integer.parseInt(month)), Year.parse(year));
72 +|             return ok(monthlyResult.orElse(MonthlyResult.empty()));
73 +|         } catch (final DateTimeParseException e) {
74 +|             final String errorMessage = String.format("The year '%s' is not a valid format", year);
75 +|             LOGGER.debug(errorMessage, e);
76 +|             LOGGER.error(errorMessage);
77 +|             return badRequest(errorMessage);
78 +|         } catch (final DateTimeException e) {
79 +|             final String errorMessage = String.format("The month '%s' is not a valid format", month);
80 +|             LOGGER.debug(errorMessage, e);
81 +|             LOGGER.error(errorMessage);
82 +|             return badRequest(errorMessage);
83 +|         } catch (final NumberFormatException e) {
84 +|             final String errorMessage = String.format("The year '%s' or month '%s' is not a valid format", year, month);
85 +|             LOGGER.debug(errorMessage, e);
86 +|             LOGGER.error(errorMessage);
87 +|             return badRequest(errorMessage);
88 +|         } catch (final Exception e) {
89 +|             LOGGER.error("Unexpected error retrieving TC result", e);
90 +|             return serverError();
91 +|         }
92 +|     }
93 +|     @PostMapping(value = "/manual/save")
94 +|     @WriteRequired
95 +|     @RequiredRoles("admin")
96 +|     public ResponseEntity<String> saveMonthlyResult() {
97 +|         LOGGER.info("GET request received to manually store monthly TC result");
98 +| 
99 +|         try {
100 +|             userStatsStorer.storeMonthlyResult();
101 +|             return ok();
102 +|         } catch (final Exception e) {
103 +|             LOGGER.error("Unexpected error manually storing TC result", e);
104 +|             return serverError();
105 +|         }
106 +|     }
107 +| }
