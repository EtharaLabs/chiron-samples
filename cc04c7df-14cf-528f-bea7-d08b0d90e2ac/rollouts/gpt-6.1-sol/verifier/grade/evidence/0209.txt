45 +| @Service
46 +| public class UserStateChangeHandler {
47 +| 
48 +|     private static final Logger LOGGER = LogManager.getLogger();
49 +|     @Autowired
50 +|     private FoldingStatsRetriever foldingStatsRetriever;
51 +| 
52 +|     @Autowired
53 +|     @Lazy
54 +|     private FoldingStatsCore foldingStatsCore;
55 +| 
56 +|     public boolean isUserStateChange(final User updatedUser, final User existingUser) {
57 +|         if (existingUser.getHardware().getId() != updatedUser.getHardware().getId()) {
58 +|             LOGGER.debug("User '{}' (ID: {}) had state change to hardware, {} -> {}", existingUser.getDisplayName(),
59 +|                 existingUser.getId(), existingUser.getHardware(), updatedUser.getHardware());
60 +|             return true;
61 +|         }
62 +| 
63 +|         if (!existingUser.getFoldingUserName().equalsIgnoreCase(updatedUser.getFoldingUserName())) {
64 +|             LOGGER.debug("User '{}' (ID: {}) had state change to Folding username, {} -> {}",
65 +|                 existingUser.getDisplayName(), existingUser.getId(), existingUser.getFoldingUserName(), updatedUser.getFoldingUserName());
66 +|             return true;
67 +|         }
68 +| 
69 +|         if (!existingUser.getPasskey().equalsIgnoreCase(updatedUser.getPasskey())) {
70 +|             LOGGER.debug("User '{}' (ID: {}) had state change to passkey, {} -> {}", existingUser.getDisplayName(),
71 +|                 existingUser.getId(), existingUser.getPasskey(), updatedUser.getPasskey());
72 +|             return true;
73 +|         }
74 +| 
75 +|         LOGGER.debug("No state change required for updated user '{}' (ID: {})", updatedUser.getDisplayName(), updatedUser.getId());
76 +|         return false;
77 +|     }
78 +| 
79 +|     public boolean isHardwareStateChange(final Hardware updatedHardware, final Hardware existingHardware) {
80 +|         // Using BigDecimal since equality checks with doubles can be imprecise
81 +|         final BigDecimal existingMultiplier = BigDecimal.valueOf(existingHardware.getMultiplier());
82 +|         final BigDecimal updatedMultiplier = BigDecimal.valueOf(updatedHardware.getMultiplier());
83 +|         final boolean isMultiplierChange = !existingMultiplier.equals(updatedMultiplier);
84 +| 
85 +|         if (isMultiplierChange) {
86 +|             LOGGER.debug("Hardware '{}' (ID: {}) had state change to multiplier, {} -> {}", updatedHardware.getId(),
87 +|                 updatedHardware.getHardwareName(), existingHardware.getMultiplier(), updatedHardware.getMultiplier());
88 +|         }
89 +| 
90 +|         return isMultiplierChange;
91 +|     }
92 +| 
93 +|     public void handleStateChange(final User userWithStateChange) {
94 +|         if (ParsingStateManager.current() == ParsingState.DISABLED) {
95 +|             LOGGER.debug("Received a state change for user '{}' (ID: {}), but system is not currently parsing stats",
96 +|                 userWithStateChange.getDisplayName(), userWithStateChange.getId());
97 +|             return;
98 +|         }
99 +| 
100 +|         try {
101 +|             final UserStats userTotalStats = foldingStatsRetriever.getTotalStats(userWithStateChange);
102 +|             LOGGER.debug("Setting initial stats to: {}", userTotalStats);
103 +|             foldingStatsCore.createInitialStats(userTotalStats);
104 +| 
105 +|             final UserTcStats currentUserTcStats = foldingStatsCore.getHourlyTcStats(userWithStateChange);
106 +|             final OffsetTcStats offsetTcStats = OffsetTcStats.create(currentUserTcStats);
107 +|             final OffsetTcStats createdOffsetStats = foldingStatsCore.createOffsetStats(userWithStateChange, offsetTcStats);
108 +|             LOGGER.debug("Added offset stats of: {}", createdOffsetStats);
109 +| 
110 +|             LOGGER.info("Handled state change for user '{}' (ID: {})", userWithStateChange.getDisplayName(), userWithStateChange.getId());
111 +|         } catch (final ExternalConnectionException e) {
112 +|             LOGGER.error("Unable to update the state of user '{}' (ID: {})", userWithStateChange.getDisplayName(), userWithStateChange.getId(), e);
113 +|         }
114 +|     }
115 +| }
