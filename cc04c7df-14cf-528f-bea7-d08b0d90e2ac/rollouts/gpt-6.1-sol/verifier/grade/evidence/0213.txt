44 +| @Service
45 +| public class UserTeamChangeHandler {
46 +| 
47 +|     private static final Logger LOGGER = LogManager.getLogger();
48 +| 
49 +|     @Autowired
50 +|     @Lazy
51 +|     private FoldingStatsCore foldingStatsCore;
52 +| 
53 +|     @Autowired
54 +|     @Lazy
55 +|     private StatsScheduler statsScheduler;
56 +| 
57 +|     public boolean isUserTeamChange(final User updatedUser, final User existingUser) {
58 +|         if (updatedUser.getTeam().getId() != existingUser.getTeam().getId()) {
59 +|             LOGGER.info("User '{}' (ID: {}) moved from team '{}' -> '{}'", existingUser.getDisplayName(), existingUser.getId(),
60 +|                 updatedUser.getTeam().getTeamName(), existingUser.getTeam().getTeamName());
61 +|             return true;
62 +|         }
63 +| 
64 +|         return false;
65 +|     }
66 +| 
67 +|     public void handleTeamChange(final User userWithTeamChange, final Team oldTeam) {
68 +|         if (ParsingStateManager.current() == ParsingState.DISABLED) {
69 +|             LOGGER.debug("Received a team change for user '{}' (ID: {}), but system is not currently parsing stats",
70 +|                 userWithTeamChange.getDisplayName(), userWithTeamChange.getId());
71 +|             return;
72 +|         }
73 +| 
74 +|         // Add user's current stats as retired stats for old team
75 +|         final UserTcStats userStats = foldingStatsCore.getHourlyTcStats(userWithTeamChange);
76 +|         final RetiredUserTcStats retiredUserTcStats =
77 +|             RetiredUserTcStats.createWithoutId(oldTeam.getId(), userWithTeamChange.getDisplayName(), userStats);
78 +|         final RetiredUserTcStats createdRetiredUserTcStats = foldingStatsCore.createRetiredUserStats(retiredUserTcStats);
79 +|         LOGGER.info("User '{}' (ID: {}) retired with retired stats ID: {}", userWithTeamChange.getDisplayName(), userWithTeamChange.getId(),
80 +|             createdRetiredUserTcStats.getRetiredUserId());
81 +| 
82 +|         // Reset user stats
83 +|         final UserStats userTotalStats = foldingStatsCore.getTotalStats(userWithTeamChange);
84 +|         foldingStatsCore.createInitialStats(userTotalStats);
85 +| 
86 +|         // Pull stats to update teams
87 +|         statsScheduler.manualTeamCompetitionStatsParsing(ProcessingType.SYNCHRONOUS);
88 +| 
89 +|         LOGGER.info("Handled team change for user '{}' (ID: {})", userWithTeamChange.getDisplayName(), userWithTeamChange.getId());
90 +|     }
91 +| }
