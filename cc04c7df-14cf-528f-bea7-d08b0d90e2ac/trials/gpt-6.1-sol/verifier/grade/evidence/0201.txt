82 +|     public Hardware createHardware(final Hardware hardware) {

88 +|     @Cached(HardwareCache.class)
89 +|     public Optional<Hardware> getHardware(final int hardwareId) {
90 +|         final Optional<Hardware> fromCache = hardwareCache.get(hardwareId);
91 +| 
92 +|         if (fromCache.isPresent()) {
93 +|             return fromCache;
94 +|         }
95 +| 
96 +|         LOGGER.trace("Cache miss! Get hardware");
97 +|         final Optional<Hardware> fromDb = dbManager.getHardware(hardwareId);
98 +|         fromDb.ifPresent(hardware -> hardwareCache.add(hardwareId, hardware));
99 +|         return fromDb;
100 +|     }
101 +| 
102 +|     @Cached(HardwareCache.class)
103 +|     public Collection<Hardware> getAllHardware() {
104 +|         final Collection<Hardware> fromCache = hardwareCache.getAll();
105 +| 
106 +|         if (!fromCache.isEmpty()) {
107 +|             return fromCache;
108 +|         }
109 +| 
110 +|         LOGGER.trace("Cache miss! Get all hardware");
111 +|         final Collection<Hardware> fromDb = dbManager.getAllHardware();
112 +| 
113 +|         for (final Hardware hardware : fromDb) {
114 +|             hardwareCache.add(hardware.getId(), hardware);
115 +|         }
116 +| 
117 +|         return fromDb;
118 +|     }
119 +| 
120 +|     @Cached({HardwareCache.class, UserCache.class})
121 +|     public Hardware updateHardware(final Hardware hardwareToUpdate) {
122 +|         final Hardware updatedHardware = dbManager.updateHardware(hardwareToUpdate);
123 +|         hardwareCache.add(updatedHardware.getId(), updatedHardware);
124 +| 
125 +|         getAllUsers()
126 +|             .stream()
127 +|             .filter(user -> user.getHardware().getId() == updatedHardware.getId())
128 +|             .map(user -> User.updateHardware(user, updatedHardware))
129 +|             .forEach(updatedUser -> userCache.add(updatedUser.getId(), updatedUser));
130 +| 
131 +|         return updatedHardware;
132 +|     }
