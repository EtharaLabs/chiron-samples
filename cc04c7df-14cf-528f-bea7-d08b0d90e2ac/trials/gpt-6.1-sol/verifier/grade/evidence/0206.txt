25 +| package me.zodac.folding.service.tc.scheduled;
26 +| 
27 +| import org.springframework.stereotype.Service;
28 +| import org.springframework.beans.factory.annotation.Autowired;
29 +| import org.springframework.context.annotation.Lazy;
30 +| 
31 +| import me.zodac.folding.state.ParsingStateManager;
32 +| import me.zodac.folding.state.SystemStateManager;
33 +| import me.zodac.folding.api.state.ParsingState;
34 +| import me.zodac.folding.api.state.SystemState;
35 +| import me.zodac.folding.service.tc.user.UserStatsResetter;
36 +| import org.apache.logging.log4j.LogManager;
37 +| import org.apache.logging.log4j.Logger;
38 +| 
39 +| @Service
40 +| public class StartOfMonthScheduler {
41 +| 
42 +|     private static final Logger LOGGER = LogManager.getLogger();
43 +| 
44 +|     @Autowired
45 +|     @Lazy
46 +|     private UserStatsResetter userStatsResetter;
47 +| 
48 +|     @org.springframework.scheduling.annotation.Scheduled(cron = "#{@schedules.start}", zone = "UTC")
49 +|     public void startOfTeamCompetition() {
50 +|         try {
51 +|     
52 +|             LOGGER.warn("Starting TC stats for new month");
53 +| 
54 +|             SystemStateManager.next(SystemState.RESETTING_STATS);
55 +|             ParsingStateManager.next(ParsingState.DISABLED);
56 +|             userStatsResetter.resetTeamCompetitionStats();
57 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
58 +|         } catch (final Exception e) {
59 +|             LOGGER.error("Error with start of team schedule", e);
60 +|         }
61 +|     }
62 +| }
