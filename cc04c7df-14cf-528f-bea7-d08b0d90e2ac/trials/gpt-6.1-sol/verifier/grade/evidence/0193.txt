25 +| package me.zodac.folding.rest.endpoint.util;
26 +| 
27 +| import org.springframework.web.bind.annotation.*;
28 +| import org.springframework.http.ResponseEntity;
29 +| 
30 +| import me.zodac.folding.rest.response.Responses;
31 +| import me.zodac.folding.api.tc.validation.ValidationResult;
32 +| 
33 +| public final class ValidationFailureResponseMapper {
34 +| 
35 +|     private ValidationFailureResponseMapper() {
36 +| 
37 +|     }
38 +| 
39 +|     public static ResponseEntity<String> map(final ValidationResult<?> validationResult) {
40 +|         switch (validationResult.getValidationFailureType()) {
41 +|             case BAD_REQUEST:
42 +|                 return Responses.badRequest(validationResult.getValidationFailure());
43 +|             case CONFLICT:
44 +|                 return Responses.conflict(validationResult.getValidationFailure());
45 +|             case NONE:
46 +|             default:
47 +|                 throw new IllegalStateException(String.format("Cannot map a REST response for validation result: %s", validationResult));
48 +|         }
49 +|     }
50 +| }
