93 +|     @GetMapping(value = "", produces = MediaType.APPLICATION_JSON_VALUE)
94 +|     @ReadRequired
95 +|     public ResponseEntity<String> getTeamCompetitionStats() {
96 +|         LOGGER.debug("GET request received to show TC stats");
97 +|

99 +|             final CompetitionSummary competitionSummary = foldingStatsCore.getCompetitionSummary();
100 +|             return ok(competitionSummary);

106 +|     @GetMapping(value = "/users/{userId}", produces = MediaType.APPLICATION_JSON_VALUE)
107 +|     @ReadRequired
108 +|     public ResponseEntity<String> getTeamCompetitionStatsForUser(@PathVariable("userId") final String userId) {
109 +| 
110 +|         try {
111 +|             final IdResult idResult = IntegerParser.parsePositive(userId);
112 +|             if (idResult.isFailure()) {
113 +|                 return idResult.getFailureResponse();
114 +|             }
115 +|             final int parsedId = idResult.getId();
116 +| 
117 +|             final Optional<User> optionalUser = foldingStatsCore.getUserWithoutPasskey(parsedId);
118 +|             if (optionalUser.isEmpty()) {
119 +|                 LOGGER.error("No user found with ID: {}", parsedId);
120 +|                 return notFound();
121 +|             }
122 +|             final User user = optionalUser.get();
123 +| 
124 +|             final CompetitionSummary competitionSummary = foldingStatsCore.getCompetitionSummary();
125 +|             final Collection<UserSummary> userSummaries = competitionSummary.getTeams()
126 +|                 .stream()
127 +|                 .flatMap(teamResult -> teamResult.getActiveUsers().stream())
128 +|                 .collect(toList());
129 +| 
130 +|             for (final UserSummary userSummary : userSummaries) {
131 +|                 if (userSummary.getUser().getId() == user.getId()) {
132 +|                     return ok(userSummary);
133 +|                 }
134 +|             }
135 +| 
136 +|             return notFound();
137 +|         } catch (final Exception e) {
138 +|             LOGGER.error("Unexpected error retrieving TC stats for users", e);
139 +|             return serverError();
140 +|         }
141 +|     }
142 +|     @PatchMapping(value = "/users/{userId}", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
143 +|     @WriteRequired
144 +|     @RequiredRoles("admin")
145 +|     public ResponseEntity<String> updateUserWithOffset(@PathVariable("userId") final String userId, @RequestBody(required = false) final OffsetTcStats offsetTcStats) {
146 +| 
147 +|         if (offsetTcStats == null || offsetTcStats.isEmpty()) {
148 +|             LOGGER.error("Payload is null");
149 +|             return nullRequest();
150 +|         }
151 +| 
152 +|         try {
153 +|             final IdResult idResult = IntegerParser.parsePositive(userId);
154 +|             if (idResult.isFailure()) {
155 +|                 return idResult.getFailureResponse();
156 +|             }
157 +|             final int parsedId = idResult.getId();
158 +| 
159 +|             final Optional<User> optionalUser = foldingStatsCore.getUserWithPasskey(parsedId);
160 +|             if (optionalUser.isEmpty()) {
161 +|                 LOGGER.error("No user found with ID: {}", parsedId);
162 +|                 return notFound();
163 +|             }
164 +| 
165 +|             final User user = optionalUser.get();
166 +|             final Hardware hardware = user.getHardware();
167 +|             final OffsetTcStats offsetTcStatsToPersist = OffsetTcStats.updateWithHardwareMultiplier(offsetTcStats, hardware.getMultiplier());
168 +|             final OffsetTcStats createdOffsetStats = foldingStatsCore.createOrUpdateOffsetStats(user, offsetTcStatsToPersist);
169 +| 
170 +|             SystemStateManager.next(SystemState.UPDATING_STATS);
171 +|             userStatsParser.parseTcStatsForUserAndWait(user);
172 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
173 +|             LOGGER.info("Updated user with ID {} with points offset: {}", userId, createdOffsetStats);
174 +|             return ok();
175 +|         } catch (final Exception e) {
176 +|             LOGGER.error("Unexpected error updating user with ID: {}", userId, e);
177 +|             return serverError();
178 +|         }
179 +|     }
