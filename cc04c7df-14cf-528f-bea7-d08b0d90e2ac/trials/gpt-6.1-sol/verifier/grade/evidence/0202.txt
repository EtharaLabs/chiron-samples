25 +| package me.zodac.folding.service.startup;
26 +| 
27 +| import org.springframework.stereotype.Service;
28 +| import org.springframework.beans.factory.annotation.Autowired;
29 +| import org.springframework.context.annotation.Lazy;
30 +| 
31 +| import java.util.Collection;
32 +| import me.zodac.folding.state.SystemStateManager;
33 +| import me.zodac.folding.api.state.SystemState;
34 +| import me.zodac.folding.api.tc.User;
35 +| import me.zodac.folding.api.tc.stats.OffsetTcStats;
36 +| import me.zodac.folding.api.tc.stats.UserStats;
37 +| import me.zodac.folding.api.util.ProcessingType;
38 +| import me.zodac.folding.service.api.FoldingStatsCore;
39 +| import me.zodac.folding.service.tc.scheduled.StatsScheduler;
40 +| import org.apache.logging.log4j.LogManager;
41 +| import org.apache.logging.log4j.Logger;
42 +| 
43 +| @Service
44 +| public class Initialiser {
45 +| 
46 +|     private static final Logger LOGGER = LogManager.getLogger();
47 +| 
48 +|     @Autowired
49 +|     @Lazy
50 +|     private FoldingStatsCore foldingStatsCore;
51 +| 
52 +|     @Autowired
53 +|     @Lazy
54 +|     private StatsScheduler statsScheduler;
55 +| 
56 +|     @org.springframework.context.event.EventListener(org.springframework.boot.context.event.ApplicationReadyEvent.class)
57 +|     public void init() {
58 +|         initCaches();
59 +|         initTcStats();
60 +| 
61 +|         SystemStateManager.next(SystemState.AVAILABLE);
62 +|         LOGGER.info("System ready for requests");
63 +|     }
64 +| 
65 +|     private void initCaches() {
66 +|         foldingStatsCore.getAllHardware();
67 +|         foldingStatsCore.getAllTeams();
68 +|         final Collection<User> users = foldingStatsCore.getAllUsersWithoutPasskeys();
69 +| 
70 +|         for (final User user : users) {
71 +|             final OffsetTcStats offsetTcStatsForUser = foldingStatsCore.getOffsetStats(user);
72 +|             LOGGER.debug("Found offset stats for user {}: {}", user, offsetTcStatsForUser);
73 +| 
74 +|             final UserStats initialStatsForUser = foldingStatsCore.getInitialStats(user);
75 +|             LOGGER.debug("Found initial stats for user {}: {}", user, initialStatsForUser);
76 +|         }
77 +| 
78 +|         LOGGER.debug("Initialised stats caches");
79 +|     }
80 +| 
81 +|     private void initTcStats() {
82 +|         if (!foldingStatsCore.isAnyHourlyTcStatsExist()) {
83 +|             LOGGER.warn("No TC stats data exists in the DB");
84 +|             statsScheduler.manualTeamCompetitionStatsParsing(ProcessingType.ASYNCHRONOUS);
85 +|         }
86 +|     }
87 +| }
