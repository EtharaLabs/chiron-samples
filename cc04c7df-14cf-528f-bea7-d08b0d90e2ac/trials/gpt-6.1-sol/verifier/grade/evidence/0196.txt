28 +|         String tag = "\"" + hash + "\"";

31 +|         int status = 200;

34 +|         ResponseEntity.BodyBuilder builder = ResponseEntity.status(status).header("Cache-Control", "no-transform, max-age=" + seconds);

40 +|     private static boolean matches(String header, String tag, boolean weak) {
41 +|         for (String part : header.split(",")) {
