83 +| 
84 +|         final ValidationResult<Team> validationResult = validateCreate(teamRequest);
85 +|         if (validationResult.isFailure()) {
86 +|             return ValidationFailureResponseMapper.map(validationResult);
87 +|         }

89 +| 
90 +|         try {
91 +|             final Team elementWithId = foldingStatsCore.createTeam(validatedTeam);

94 +|                 .pathSegment(String.valueOf(elementWithId.getId()));
95 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
96 +|             LOGGER.info("Created team with ID {}", elementWithId.getId());
97 +|             return created(elementWithId, elementLocationBuilder);

99 +|             LOGGER.error("Unexpected error creating team: {}", teamRequest, e);

103 +|     @PostMapping(value = "/batch", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
104 +|     @WriteRequired
105 +|     @RequiredRoles("admin")
106 +|     public ResponseEntity<String> createBatchOf(@RequestBody(required = false) final Collection<TeamRequest> teamRequests) {
107 +| 
108 +|         final Collection<Team> validTeams = new ArrayList<>(teamRequests.size() / 2);
109 +|         final Collection<ValidationFailure> failedValidationResponses = new ArrayList<>(teamRequests.size() / 2);
110 +| 
111 +|         for (final TeamRequest teamRequest : teamRequests) {
112 +|             final ValidationResult<Team> validationResult = validateCreate(teamRequest);
113 +| 
114 +|             if (validationResult.isFailure()) {
115 +|                 LOGGER.error("Found validation error for {}: {}", teamRequest, validationResult);
116 +|                 failedValidationResponses.add(validationResult.getValidationFailure());
117 +|             } else {
118 +|                 validTeams.add(validationResult.getOutput());
119 +|             }
120 +|         }
121 +| 
122 +|         if (validTeams.isEmpty()) {
123 +|             LOGGER.error("All teams contain validation errors: {}", failedValidationResponses);
124 +|             return badRequest(failedValidationResponses);
125 +|         }
126 +| 
127 +|         final List<Object> successful = new ArrayList<>();
128 +|         final List<Object> unsuccessful = new ArrayList<>(failedValidationResponses);
129 +| 
130 +|         for (final Team validTeam : validTeams) {
131 +|             try {
132 +|                 final Team teamWithId = foldingStatsCore.createTeam(validTeam);
133 +|                 successful.add(teamWithId);
134 +|             } catch (final Exception e) {
135 +|                 LOGGER.error("Unexpected error creating team: {}", validTeam, e);
136 +|                 unsuccessful.add(validTeam);
137 +|             }
138 +|         }
139 +| 
140 +|         final BatchCreateResponse batchCreateResponse = BatchCreateResponse.create(successful, unsuccessful);
141 +| 
142 +|         if (successful.isEmpty()) {
143 +|             return badRequest(batchCreateResponse);
144 +|         }
145 +| 
146 +|         if (!unsuccessful.isEmpty()) {
147 +|             LOGGER.error("{} teams successfully created, {} teams unsuccessful", successful.size(), unsuccessful.size());
148 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
149 +|             return ok(batchCreateResponse);
150 +|         }
151 +| 
152 +|         LOGGER.info("{} teams successfully created", successful.size());
153 +|         SystemStateManager.next(SystemState.WRITE_EXECUTED);
154 +|         return ok(batchCreateResponse.getSuccessful());
155 +|     }

220 +|     @PutMapping(value = "/{teamId}", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
221 +|     @WriteRequired
222 +|     @RequiredRoles("admin")
223 +|     public ResponseEntity<String> updateById(@PathVariable("teamId") final String teamId, @RequestBody(required = false) final TeamRequest teamRequest) {
224 +| 
225 +|         if (teamRequest == null) {
226 +|             LOGGER.error("No payload provided");
227 +|             return nullRequest();
228 +|         }
229 +| 
230 +|         try {
231 +|             final IdResult idResult = IntegerParser.parsePositive(teamId);
232 +|             if (idResult.isFailure()) {
233 +|                 return idResult.getFailureResponse();
234 +|             }
235 +|             final int parsedId = idResult.getId();
236 +| 
237 +|             final Optional<Team> optionalElement = foldingStatsCore.getTeam(parsedId);
238 +|             if (optionalElement.isEmpty()) {
239 +|                 LOGGER.error("No team found with ID {}", teamId);
240 +|                 return notFound();
241 +|             }
242 +|             final Team existingTeam = optionalElement.get();
243 +| 
244 +|             if (existingTeam.isEqualRequest(teamRequest)) {
245 +|                 LOGGER.debug("No change necessary");
246 +|                 return ok(existingTeam);
247 +|             }
248 +| 
249 +|             final ValidationResult<Team> validationResult = validateUpdate(teamRequest, existingTeam);
250 +|             if (validationResult.isFailure()) {
251 +|                 return ValidationFailureResponseMapper.map(validationResult);
252 +|             }
253 +|             final Team validatedHardware = validationResult.getOutput();
254 +| 
255 +|             // The payload 'should' have the ID, but it's not guaranteed if the correct URL is used
256 +|             final Team teamWithId = Team.updateWithId(existingTeam.getId(), validatedHardware);
257 +|             final Team updatedTeamWithId = foldingStatsCore.updateTeam(teamWithId);
258 +| 
259 +|             final UriComponentsBuilder elementLocationBuilder = ServletUriComponentsBuilder.fromCurrentRequest()
260 +|                 .pathSegment(String.valueOf(updatedTeamWithId.getId()));
261 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
262 +|             LOGGER.info("Updated team with ID {}", updatedTeamWithId.getId());
263 +|             return ok(updatedTeamWithId, elementLocationBuilder);
264 +|         } catch (final Exception e) {
265 +|             LOGGER.error("Unexpected error updating team with ID: {}", teamId, e);
266 +|             return serverError();
267 +|         }
268 +|     }
