27 +| import org.springframework.web.bind.annotation.*;
28 +| 
29 +| import static me.zodac.folding.rest.response.Responses.badRequest;
30 +| 
31 +| import org.apache.logging.log4j.LogManager;
32 +| import org.apache.logging.log4j.Logger;
33 +| 
34 +| public final class IntegerParser {
35 +| 
36 +|     private static final Logger LOGGER = LogManager.getLogger();
37 +| 
38 +|     private IntegerParser() {
39 +| 
40 +|     }
41 +| 
42 +|     public static IdResult parsePositive(final String integer) {
43 +|         try {
44 +|             final int parsedId = Integer.parseInt(integer);
45 +|             if (parsedId < 0) {
46 +|                 final String errorMessage = String.format("The ID '%s' is out of range", parsedId);
47 +|                 LOGGER.error(errorMessage);
48 +|                 return IdResult.failure(badRequest(errorMessage));
49 +|             }
50 +| 
51 +|             return IdResult.success(parsedId);
52 +|         } catch (final NumberFormatException e) {
53 +|             final String errorMessage = String.format("The ID '%s' is not a valid format", integer);
54 +|             LOGGER.error(errorMessage);
55 +|             return IdResult.failure(badRequest(errorMessage));
56 +|         }
57 +|     }
58 +| }
