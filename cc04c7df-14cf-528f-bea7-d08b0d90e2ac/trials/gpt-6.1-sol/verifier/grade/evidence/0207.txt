25 +| package me.zodac.folding.service.tc.scheduled;
26 +| 
27 +| import org.springframework.stereotype.Service;
28 +| import org.springframework.beans.factory.annotation.Autowired;
29 +| import org.springframework.context.annotation.Lazy;
30 +| 
31 +| import java.util.Collection;
32 +| import me.zodac.folding.api.state.ParsingState;
33 +| import me.zodac.folding.api.state.SystemState;
34 +| import me.zodac.folding.api.tc.Team;
35 +| import me.zodac.folding.api.tc.User;
36 +| import me.zodac.folding.api.util.ProcessingType;
37 +| import me.zodac.folding.service.api.FoldingStatsCore;
38 +| import me.zodac.folding.service.tc.user.UserStatsParser;
39 +| import me.zodac.folding.state.ParsingStateManager;
40 +| import me.zodac.folding.state.SystemStateManager;
41 +| import org.apache.logging.log4j.LogManager;
42 +| import org.apache.logging.log4j.Logger;
43 +| 
44 +| @Service
45 +| public class StatsScheduler {
46 +| 
47 +|     private static final Logger LOGGER = LogManager.getLogger();
48 +| 
49 +|     // Default is to run every hour at 55 minutes past the hour
50 +| 
51 +|     @Autowired
52 +|     @Lazy
53 +|     private FoldingStatsCore foldingStatsCore;
54 +| 
55 +|     @Autowired
56 +|     @Lazy
57 +|     private UserStatsParser userStatsParser;
58 +| 
59 +|     @org.springframework.scheduling.annotation.Scheduled(cron = "#{@schedules.stats}", zone = "UTC")
60 +|     public void scheduledTeamCompetitionStatsParsing() {
61 +| 
62 +|         try {
63 +|             ParsingStateManager.next(ParsingState.ENABLED_TEAM_COMPETITION);
64 +|             SystemStateManager.next(SystemState.UPDATING_STATS);
65 +|             parseTeamCompetitionStats(ProcessingType.ASYNCHRONOUS);
66 +|             SystemStateManager.next(SystemState.WRITE_EXECUTED);
67 +|         } catch (final Exception e) {
68 +|             LOGGER.error("Unexpected error updating TC stats", e);
69 +|         }
70 +|     }
71 +| 
72 +|     public void manualTeamCompetitionStatsParsing(final ProcessingType processingType) {
73 +|         LOGGER.debug("Manual stats parsing execution");
74 +|         ParsingStateManager.next(ParsingState.ENABLED_TEAM_COMPETITION);
75 +|         SystemStateManager.next(SystemState.UPDATING_STATS);
76 +|         parseTeamCompetitionStats(processingType);
77 +|         SystemStateManager.next(SystemState.WRITE_EXECUTED);
78 +|     }
79 +| 
80 +|     private void parseTeamCompetitionStats(final ProcessingType processingType) {
81 +|         LOGGER.info("");
82 +|         LOGGER.info("Parsing TC Folding stats:");
83 +| 
84 +|         final Collection<Team> tcTeams = foldingStatsCore.getAllTeams();
85 +|         if (tcTeams.isEmpty()) {
86 +|             LOGGER.warn("No TC teams configured in system!");
87 +|             return;
88 +|         }
89 +| 
90 +|         LOGGER.debug("Found TC teams: {}", tcTeams);
91 +|         for (final Team team : tcTeams) {
92 +|             parseTcStatsForTeam(team, processingType);
93 +|         }
94 +|     }
95 +| 
96 +|     private void parseTcStatsForTeam(final Team team, final ProcessingType processingType) {
97 +|         LOGGER.debug("Getting TC stats for users in team {}", team::getTeamName);
98 +|         final Collection<User> teamUsers = foldingStatsCore.getUsersOnTeamWithPasskeys(team);
99 +| 
100 +|         if (teamUsers.isEmpty()) {
101 +|             LOGGER.warn("No users for team '{}'", team.getTeamName());
102 +|             return;
103 +|         }
104 +| 
105 +|         LOGGER.debug("Found users TC team {}: {}", team.getTeamName(), teamUsers);
106 +|         for (final User user : teamUsers) {
107 +|             if (processingType == ProcessingType.ASYNCHRONOUS) {
108 +|                 userStatsParser.parseTcStatsForUser(user);
109 +|             } else {
110 +|                 userStatsParser.parseTcStatsForUserAndWait(user);
111 +|             }
112 +|         }
113 +|     }
114 +| }
